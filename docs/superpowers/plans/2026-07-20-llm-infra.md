```ts
// apps/api/src/modules/llm/sse-parser.ts

export interface ParsedSseEvent {
  /** Raw contents of the `data:` field (may be JSON or `[DONE]`). */
  data: string;
}

/**
 * Incremental SSE parser for OpenAI-compatible chat streams.
 * Feed successive byte chunks; carry `rest` between reads.
 */
export function parseSseChunk(buffer: string): { events: ParsedSseEvent[]; rest: string } {
  const events: ParsedSseEvent[] = [];
  let start = 0;

  while (true) {
    const sep = buffer.indexOf('\n\n', start);
    if (sep === -1) {
      break;
    }
    const block = buffer.slice(start, sep);
    start = sep + 2;

    const lines = block.split('\n');
    const dataLines: string[] = [];
    for (const line of lines) {
      if (line.startsWith(':')) {
        continue; // keep-alive / comment
      }
      if (line.startsWith('data:')) {
        // Spec allows optional space after colon
        dataLines.push(line.slice(5).replace(/^\s/, ''));
      }
    }
    if (dataLines.length > 0) {
      events.push({ data: dataLines.join('\n') });
    }
  }

  return { events, rest: buffer.slice(start) };
}
```

### Env resolution (implement exactly)

| Var | Default |
|---|---|
| `LLM_BASE_URL` | `http://localhost:11434/v1` |
| `LLM_API_KEY` | empty |
| `LLM_CONTENT_MODEL` | `qwen3:1.7b` |
| `LLM_ASSISTANT_MODEL` | `qwen3:4b-instruct` |
| `LLM_TIMEOUT_MS` | `30000` |
| `LLM_CONTENT_ENABLED` | `false` (read by content-gen, not LlmService) |

Model resolution: `opts.model` wins; else `purpose === 'assistant'` → `LLM_ASSISTANT_MODEL`; else `LLM_CONTENT_MODEL`.

Auth header: set `Authorization: Bearer ${key}` **only when** `LLM_API_KEY` is non-empty after trim.

### 429 backoff (inside `chat` / `chatStream` request loop)

- Max **3** attempts total (initial + 2 retries).
- On HTTP 429: parse `Retry-After` as seconds → ms; if missing/invalid use **1000ms**.
- If wait would exceed remaining `timeoutMs` budget → throw `LlmRateLimitError`.
- Else sleep, retry.
- Other non-2xx → `LlmResponseError` immediately (no retry).
- Network failure → `LlmUnavailableError`.
- Abort/timeout → `LlmTimeoutError`.

### Usage logging

After every terminal outcome (success or throw path), call `LlmUsageService.log` fail-soft with purpose/model/tokens/latency/success/error. Never let logging break the caller.

```ts
export class LlmError extends Error {
  readonly code: string;
  constructor(message: string, code: string) {
    super(message);
    this.name = new.target.name;
    this.code = code;
  }
}

export class LlmUnavailableError extends LlmError {
  constructor(message = 'LLM provider unreachable') {
    super(message, 'llm_unavailable');
  }
}

export class LlmTimeoutError extends LlmError {
  constructor(message = 'LLM request timed out') {
    super(message, 'llm_timeout');
  }
}

export class LlmResponseError extends LlmError {
  readonly status?: number;
  constructor(message: string, status?: number) {
    super(message, 'llm_response');
    this.status = status;
  }
}

export class LlmRateLimitError extends LlmError {
  readonly retryAfterMs?: number;
  constructor(message = 'LLM rate limited', retryAfterMs?: number) {
    super(message, 'llm_rate_limit');
    this.retryAfterMs = retryAfterMs;

}
```

```ts
// apps/api/src/modules/llm/llm-usage.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { LlmUsagePurpose } from '@repo/shared';

import { DatabaseService } from '../../common/database/database.service';

export interface LlmUsageLogParams {
  userId?: string;
  purpose: LlmUsagePurpose;
  model: string;
  promptTokens?: number;
  completionTokens?: number;
  latencyMs?: number;
  success: boolean;
  error?: string;
}

@Injectable()
export class LlmUsageService {
  private readonly logger = new Logger(LlmUsageService.name);

  constructor(private readonly databaseService: DatabaseService) {}

  /** Insert one usage row. Failures never break the caller. */
  async log(params: LlmUsageLogParams): Promise<void> {
    try {
      await this.databaseService.query(
        `INSERT INTO llm_usage_log
           (user_id, purpose, model, prompt_tokens, completion_tokens, latency_ms, success, error)
         VALUES ($1, $2, $3, $4, $5, $6, $7, $8)`,
        [
          params.userId ?? null,
          params.purpose,
          params.model,
          params.promptTokens ?? null,
          params.completionTokens ?? null,
          params.latencyMs ?? null,
          params.success,
          params.error ? params.error.slice(0, 200) : null,
        ]
      );
    } catch (error: unknown) {
      this.logger.error(
        `Failed to log LLM usage: ${error instanceof Error ? error.message : String(error)}`
      );
    }
  }
}
```

```ts
export class ContentGenerationService {

  isEnabled(): boolean {

    return v === 'true' || v === '1' || v === 'yes';
  }

  async rewriteTitle(input: ContentRewriteInput): Promise<string> {
    if (!this.isEnabled()) return input.baseTitle;
    // build system + user messages from existing prompt lines
    try {
      const { text } = await this.llm.chat(messages, {
        purpose: 'content',
        temperature: 0.3,

      );
    }
```
